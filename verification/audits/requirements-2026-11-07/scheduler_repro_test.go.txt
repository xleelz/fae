package scheduler
import("context";"testing";"time";"github.com/xleelz/fae/hostguard")
func TestAuditPriorityIsReversed(t *testing.T){l,o,c:=slotFixture(t);c.Sample.Memory.Value=10*float64(hostguard.GiB);high:=slotHolder("high",o);high.Priority=100;high.Memory=8*hostguard.GiB;high.Ready=time.Now().Add(-time.Hour);d,e:=l.AcquireWorkflowSlot(context.Background(),high,c);if e!=nil||d.Granted{t.Fatal(d,e)};low:=slotHolder("low",o);low.Priority=10;d,e=l.AcquireWorkflowSlot(context.Background(),low,c);if e!=nil{t.Fatal(e)};if !d.Granted{t.Fatalf("bug no longer reproduces: %+v",d)};t.Log("CONFIRMED: priority10 granted ahead of queued priority100")}
func TestAuditMoleculeExemptionStaysAfterLimitClears(t *testing.T){l,o,c:=slotFixture(t);ctx:=context.Background();c.Sample.Memory.Value=14*float64(hostguard.GiB);held:=slotHolder("held",o);held.Scope="molecule";held.Workflow="same";d,e:=l.AcquireWorkflowSlot(ctx,held,c);if e!=nil||!d.Granted{t.Fatal(d,e)};ahead:=slotHolder("ahead",o);ahead.Scope="molecule";ahead.Workflow="same";ahead.MoleculeLimit=1;ahead.Priority=10;ahead.Ready=time.Now().Add(-time.Hour);d,e=l.AcquireWorkflowSlot(ctx,ahead,c);if e!=nil||d.Reason!="molecule_limit"{t.Fatal(d,e)};behind:=slotHolder("behind",o);behind.Instance="other";behind.Priority=10;behind.Memory=8*hostguard.GiB;d,e=l.AcquireWorkflowSlot(ctx,behind,c);if e!=nil||d.Granted{t.Fatal(d,e)};if e=l.ReleaseWorkflowSlot(ctx,held.ID,o);e!=nil{t.Fatal(e)};d,e=l.AcquireWorkflowSlot(ctx,behind,c);if e!=nil||!d.Granted{t.Fatal("bug no longer reproduces",d,e)};_,state,e:=l.WorkflowSlotSnapshot(ctx,c);if e!=nil{t.Fatal(e)};if !state.Waiters[ahead.ID].IgnoreFIFO{t.Fatal("missing stale exemption")};t.Log("CONFIRMED: now-eligible ahead waiter still IgnoreFIFO; behind owner acquired")}
func TestAuditSmallerRevisionLeavesObsoleteResizeAtQueueHead(t *testing.T){l,o,c:=slotFixture(t);ctx:=context.Background();root:=slotHolder("root",o);root.Priority=10;root.Ready=time.Now().Add(-time.Hour);d,e:=l.AcquireWorkflowSlot(ctx,root,c);if e!=nil||!d.Granted{t.Fatal(d,e)};c.Sample.Memory.Value=10*float64(hostguard.GiB);large:=root;large.Revision=2;large.Memory=12*hostguard.GiB;d,e=l.AcquireWorkflowSlot(ctx,large,c);if e!=nil||d.Granted{t.Fatal(d,e)};smaller:=root;smaller.Revision=3;smaller.Memory=2*hostguard.GiB;d,e=l.AcquireWorkflowSlot(ctx,smaller,c);if e!=nil||!d.Granted||!d.Repriced{t.Fatal(d,e)};c.Sample.Memory.Value=20*float64(hostguard.GiB);other:=slotHolder("other",o);other.Ready=time.Now();d,e=l.AcquireWorkflowSlot(ctx,other,c);if e!=nil{t.Fatal(e)};_,state,e:=l.WorkflowSlotSnapshot(ctx,c);if e!=nil{t.Fatal(e)};if d.Granted||state.Waiters[root.ID].Revision!=2{t.Fatal("bug no longer reproduces",d,state)};t.Log("CONFIRMED: live revision3 fits its granted reservation, obsolete revision2 resize remains FIFO head and blocks another fitting holder")}
