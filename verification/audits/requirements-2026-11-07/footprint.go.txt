package main
import("fmt";"os";"strconv";"runtime";"time";"encoding/json";"github.com/xleelz/fae/workflow")
func main(){n,_:=strconv.Atoi(os.Args[1]);d:=workflow.Document{ID:"audit",SlotScope:"tree",Start:"s0",Stages:map[string]workflow.Stage{}};for i:=0;i<n;i++{id:=fmt.Sprint("s",i);next:="Done";if i+1<n{next=fmt.Sprint("s",i+1)};d.Stages[id]=workflow.Stage{Primitive:workflow.AgentTurn,Role:"author"};d.Edges=append(d.Edges,workflow.Edge{From:id,To:next})};raw,_:=json.Marshal(d);runtime.GC();var before,after runtime.MemStats;runtime.ReadMemStats(&before);start:=time.Now();f,e:=workflow.DeriveFootprint(d,nil,0);runtime.ReadMemStats(&after);fmt.Printf("stages=%d json_bytes=%d derived=%d err=%v time=%v allocated_MiB=%.1f heap_sys_MiB=%.1f\n",n,len(raw),f.DerivedSessions,e,time.Since(start),float64(after.TotalAlloc-before.TotalAlloc)/(1<<20),float64(after.HeapSys)/(1<<20))}
