package main
import("fmt";"encoding/json";"github.com/xleelz/fae/workflow";"github.com/xleelz/fae/engine";"github.com/xleelz/fae/domain")
func doc(id,next string,budget bool)workflow.Document{d:=workflow.Document{SchemaVersion:2,ID:id,Start:"call",SlotScope:"tree",MaxSessions:1,TargetRequirements:&workflow.TargetRequirements{},Limits:workflow.Limits{Attempts:20,TestRuns:1,ScopeWidens:1},Roles:map[string]workflow.Role{"author":{Instructions:"unused",Briefs:map[string]string{"default":"unused"},Routes:[]workflow.Binding{{Provider:"script",Model:"recorded"}},Scope:"task-bound",Cap:1,Outputs:[]string{"Contract"}}},Artifacts:map[string]workflow.ArtifactType{"Contract":{Version:1,Schema:json.RawMessage(`{"type":"object"}`)}},Stages:map[string]workflow.Stage{"call":{Primitive:workflow.SubWorkflow,Workflow:&workflow.WorkflowRef{ID:next,Revision:1},Output:"Contract",Failure:"stop",OutputMap:[]workflow.ArtifactMap{{Name:"contract",Stage:"call",Type:"Contract"}}},"stop":{Primitive:workflow.Escalate,Kind:domain.HumanDecisionRequired,Reason:"child failed"}},Edges:[]workflow.Edge{{From:"call",To:"Done"},{From:"call",To:"stop"}}};if budget{d.Recursion=&workflow.Recursion{MaxDepth:8,MaxDescendants:20,MaxChildrenPerVisit:1}};return d}
func main(){a:=doc("a","b",true);b:=doc("b","c",true);c:=doc("c","d",false);d:=doc("d","a",true);for _,v:=range []workflow.Document{a,b,c,d}{fmt.Println("validation",v.ID,workflow.ValidateWrite(v))};s,e:=engine.SimulateScenario(a,"root",workflow.Scenario{Workflows:map[string]workflow.Document{"a":a,"b":b,"c":c,"d":d}},workflow.SimulationOptions{});fmt.Println("err",e,"state",s.State);for _,n:=range s.Tree{fmt.Println("node",n.Depth,n.Workflow,n.Task,n.Reason)};j,_:=engine.New("filter-proof",1,a,"tree");for n:=0;n<5&&j.Child==nil;n++{j,_,_,_=engine.Prepare(j)};j,events,err:=engine.ChildFailed(j,"recursion_budget: max_descendants exceeded");fmt.Println("child-failed",err,"history",len(j.History));for _,ev:=range events{fmt.Printf("actual_event kind=%q cause=%q tree_filter_matches=%v\n",ev.Kind,ev.Cause,ev.Kind=="recursion-budget"||ev.Cause=="recursion-budget")}}
